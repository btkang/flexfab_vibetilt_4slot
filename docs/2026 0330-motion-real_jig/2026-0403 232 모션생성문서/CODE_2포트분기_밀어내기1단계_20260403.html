4. 코드 변경 상세

MotionJig.cs — uart_id param 오버라이드

// VL 센서 (232 or 485, param의 uart_id로 오버라이드 가능)
string vlKey = isRs485 ? "uart_485" : "uart_232";
if (param.TryGetValue("uart_id", out var uartIdObj) && uartIdObj != null)
    vlKey = uartIdObj.ToString();

VibeTilt.cs — ResolveUart() uart_id 우선 적용

// param에서 uart_id로 오버라이드 가능 (밀어내기식 2포트 지원)
if (param.TryGetValue("uart_id", out var uartIdOverride) &&
    uartIdOverride != null &&
    !string.IsNullOrWhiteSpace(uartIdOverride.ToString()))
{
    uartId = uartIdOverride.ToString();
}
