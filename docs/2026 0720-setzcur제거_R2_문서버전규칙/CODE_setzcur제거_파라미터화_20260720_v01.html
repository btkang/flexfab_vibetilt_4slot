// 6. 자이로 기준점 설정 (setzcur) — 기본 미실행

// 제거 적용 전 센서에 setzfac(공장영점 복원) 1회 선행 필수.
// 되돌리기: JSON #3/#10 param에 "gyro_setzcur": true
if (GetParamBool(param, "gyro_setzcur", false))
{
    if (!GyroSetZero(gyro, log)) { ...기존 FAIL 처리... }
